\section{Metric equational system}

\begin{frame}{The equational system: a first glimpse}

        $t =_\epsilon s$ reads as ``$t$ and $s$ are \emph{at most}
        at distance $\epsilon$"

        \bigskip
        Metric generalisations of eq. laws emerge; others 
        become apparent  

        \bigskip
        \begin{examples}
                \[
                        \infer[(trans)]{t =_{q + r} u}
                        {t =_q s \qquad s =_r u}
                        \hspace{2cm}
                        \infer[(weak)]{t =_r s}
                        {t =_q s \quad q \leq r} 
                \]
                \[
                        \infer[(refl)]{t =_0 t}{}
                \]
        \end{examples}

\end{frame}

\begin{frame}{For a truly complete system for lambda-calculus \dots}
        We need to
        \begin{itemize}
                \item integrate metric equational laws
                        \\[5pt]
                \item integrate system of Lecture 1 
                        \\[5pt]
                \item dictate how metrics interact with term constructs
        \end{itemize}
\end{frame}

\begin{frame}{Metric equational laws}

\end{frame}

\begin{frame}{Interaction between metric and term constructs}
    \begin{figure}[h!]
    \begin{flalign*}
        \infer[]{
                t \ \text{to}\ \ast.\ u =_{q \otimes r}
                s \ \text{to}\ \ast.\ v 
        }
        {       
                t =_q s \qquad u =_r v
        }
        \hspace{0.8cm}
        \infer[]{
                \text{pm}\ t\ \text{to}\ x \otimes y.\ u =_{q \otimes r}
                \text{pm}\ s\ \text{to}\ x \otimes y.\ v 
        }{
                t =_q s \qquad u =_r v
        }
    \end{flalign*}
    \begin{flalign*}
        \infer[]{
                t \otimes u =_{q \otimes r} s \otimes v 
        }
        {       
                t =_q s \qquad u =_r v  
        }
        \hspace{1cm}
        \infer[]{
                \lambda x. t =_q  \lambda x.  s
        }{
                t =_q s
        }
        \hspace{1cm}
        \infer[]{
                t \, u =_{q \otimes r} s \, v
        }{
                t =_q s \qquad u =_r v
        }
    \end{flalign*} 
\end{figure}  
\end{frame}

\begin{frame}{Reasoning example}

        \begin{block}{Quantitative extensionality}
        \[
                t \> x =_q s \> x \, \Longleftrightarrow \,  t =_q s
        \]
        \end{block}

        \pause
        \bigskip
        Proof sketch
        \begin{flalign*}
                t \> x =_q s \> x 
                & \Longrightarrow
                \lambda x. \, t \> x =_q \lambda x. \, s \> x
                \\
                & \Longrightarrow
                t =_q s 
                \\
                & \Longrightarrow
                t \> x =_q s \> x
        \end{flalign*}
\end{frame}

\section{Categorical semantics}

\begin{frame}{Main idea}

        $t = s \, \, \, \Longrightarrow \sem{t} = \sem{s}$

        \bigskip
        \pause
        $t =_q s \Longrightarrow \dots$

        \bigskip
        \pause
        The sets of arrows need to be \alert{metric spaces}

        \bigskip
        $t =_q s \Longrightarrow d(\sem{t},\sem{s}) \leq q$
\end{frame}

\begin{frame}{Main idea}
        \dots\ also all categorical operations associated
        to term constructs need to be \alert{non-expansive}, viz.

        \begin{itemize}
                \item $\catC(A,B) \otimes \catC(B,C) \to \catC(A,C)$
                        \hspace{2cm} (composition) 
                \item $\catC(A,B) \otimes \catC(C,D) \to \catC(A \otimes C, B \otimes D)$
                        \hspace{0.57cm} (tensoring) 
                \item $\catC(A \otimes B, C) \to \catC(A, B \multimap C)$
                        \hspace{2.01cm} (currying) 
        \end{itemize}

        \pause
        \bigskip
        We call such categories \alert{metrically enriched}
\end{frame}

\begin{frame}{Exercise}

        Show that the category of metric spaces and non-expansive
        functions is metrically enriched
\end{frame}
