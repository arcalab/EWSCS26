\section{Theories, soundness, and completeness}

\begin{frame}{Theory}

        \begin{definition}[Signature]
                Set of basic types + operation symbols
                $f : \typeA_1, \dots, \typeA_n \to \typeB$
        \end{definition}


        \bigskip
        \pause
        \begin{definition}[Theory]
                A signature + a set of basic metric equations (axioms)
        \end{definition}


        \bigskip
        \pause
        \begin{definition}[Theorem]
                A metric equation that is provable in a given theory
        \end{definition}
\end{frame}

\begin{frame}{Soundness}

        Pick a theory and any associated theorem $t =_q s$
        
        Pick an interpretation $\sem{-}$ 

        If all axioms are satisfied by the interpretation (a model) then
        \[
                d(\sem{t},\sem{s}) \leq q
        \]
                
        \pause
        \bigskip
        Proof sketch: routine induction over proof derivation trees
\end{frame}


\begin{frame}{Completeness}

        Pick a theory $\mathcal{T}$ and suppose all models satisfy
        \[
                d(\sem{t},\sem{s}) \leq q
        \]

        \medskip
        For every \alert{approximation} $q + \epsilon > q$
        \[
                t =_{q + \epsilon} s \text{ is a theorem }
        \]

        \pause
        \bigskip
        Proof idea: build a metrically enriched smc category
        from $\mathcal{T}$

        Then show $d(t,s) \leq q \Longrightarrow t =_{q + \epsilon} s$ is a theorem
\end{frame}

\begin{frame}{The syntactic category}

        \begin{itemize}
                \item Objects are the types
                \item Morphisms $A \to B$ ``are"
                        the terms $x : A \vdash t : B$
                \item $d(t,s) = \bigwedge
                        \Big \{ q \mid t =_q s \> \text{ is a theorem } \Big \}$
        \end{itemize}

        \medskip
        \begin{align*}
                d(t,s) = & \bigwedge
                        \Big \{ r \mid t =_r s \> \text{ is a theorem } \Big \}
                        \leq q < q + \epsilon
                        \\
                         &
                         \Longrightarrow
                         \exists r. \,
                         r \leq q + \epsilon \> \wedge \>
                         t =_r s \text{ is a theorem }
                         \\
                         &
                         \Longrightarrow
                         t =_{q + \epsilon} s \text{ is a theorem }
        \end{align*}
\end{frame}

\begin{frame}{Other type constructs -- conditionals (disjunction)}

        \[
                \infer{
                        \Gamma, \Delta \vdash
                        \text{case} \>  t \> \{ \, x \Rightarrow s ; \> y \Rightarrow u \} 
                        : \typeC
                }{
                        \Gamma \vdash t : \typeA \oplus \typeB
                        \qquad
                        \Delta, x : \typeA \vdash s : \typeC
                        \qquad
                        \Delta, y : \typeB \vdash u : \typeC
                }
        \]

        \pause
        \bigskip
        \[
                \infer{
                        \text{case} \>  t \> \{ \, x \Rightarrow s ; \> y \Rightarrow u \, \} 
                        =_{q \vee r}
                        \text{case} \>  t \> \{ \, x \Rightarrow v ; \> y \Rightarrow w \, \} 
                }{
                        s =_q v \qquad u =_r w
                }
        \]
\end{frame}

\section{Application to probabilistic programming}

\begin{frame}{Banach spaces (linear algebra for prob. programming)}

        Vector space $V$ with a notion of size (norm)
        \[
              \lVert - \rVert : V \longrightarrow \Rz
        \]
        plus closure under limits

        \bigskip
        \begin{examples}
         The real numbers $\Reals$, continuous distributions $M(\Reals)$ \dots
        \end{examples}

        \pause
        \bigskip
        The function $(v_1,v_2) \mapsto \lVert v_1 - v_2 \rVert$ is a \alert{metric}
\end{frame}

\begin{frame}{Continuous distributions over the reals}

        These are functions
        \[
                \mu : \{ \text{opens of } \Reals \} \longrightarrow \Rz
        \]
        plus some coherence conditions


        \bigskip
        \pause
        We also have the norm
        \[
                \lVert \mu \rVert = \sup
                \left \{ \sum_{i = 1}^n | \mu(U_i) | \mid \{U_1,\dots,U_n\} \text{ a partition of opens} \right \}
        \]

        \pause
        which can be cumbersome to calculate \dots
\end{frame}

\begin{frame}{Linear structure over Banach spaces}

        Norms in tensor products $V \otimes W$ also cumbersome to calculate

        \medskip
        $V \multimap W$ = $(\{ \text{continuous linear maps } V \to W \},\lVert - \rVert)$
        \[
          \lVert f \rVert = \sup \{ \lVert f (v) \rVert \mid \lVert v \rVert \leq 1 \}      
        \]
\end{frame}

\begin{frame}{A probabilistic theory}

        \begin{itemize}
                \item $\sem{\mathtt{reals}} = M(\Reals)$ 
                \item $\sem{+ : \mathtt{reals},\mathtt{reals} \to \mathtt{reals}} : M(\Reals) \otimes M(\Reals) 
                                \to M(\Reals)$
                \item $\sem{\mathtt{unif}(a,b) : \typeI \to \mathtt{reals}} : \Reals \to M(\Reals)$
        \end{itemize}

        \bigskip
        plus a simple axiom
        \[
                \mathtt{unif}(0,1) =_{\frac{2 \epsilon}{1 + \epsilon}}
                \mathtt{unif}(0,1+\epsilon)
        \]
\end{frame}

\begin{frame}{A probabilistic theory}
        \[
                \mathtt{unif}(0,1) =_{\frac{2 \epsilon}{1 + \epsilon}}
                \mathtt{unif}(0,1+\epsilon)
        \]
        \bigskip
        \centering
        \includegraphics[scale=0.4]{../images/distance.png}
\end{frame}

\begin{frame}{Random walks}
        Basic ingredients

        \medskip
        \[
                \infer{
                        (-) \vdash \underbrace{ \lambda x. \, x + \mathtt{unif}(0,a)}_{\mathtt{jump}_a} 
                        : \mathtt{real} \multimap \mathtt{real}
                }{
                        x : \mathtt{real} \vdash x + \mathtt{unif}(0,1)
                        : \mathtt{real}
                }
        \]
        
        \[
                x : \mathtt{real} \vdash
                \underbrace{\lambda f_1 \> \dots \> \lambda f_n . \> f_1 \dots f_n (x)}_{\mathtt{seq}_n(x)}
        \]
\end{frame}

\begin{frame}{Random walks}

        We can then derive
        \begin{align*}
                x : \mathtt{real} \vdash 
                & \> \mathtt{seq}_n(x) \>
                \mathtt{jump}_1 \dots \mathtt{jump}_1
                \\
                & =_{ n \frac{2\epsilon}{1 + \epsilon} }
                \\
                & \> \mathtt{seq}_n(x) \>
                \mathtt{jump}_{1 + \epsilon} \dots \mathtt{jump}_{1 + \epsilon}
                : \mathtt{real}
        \end{align*}
\end{frame}

\begin{frame}{Two dimensional random walks}

        \dots\ or for example
        \begin{align*}
                x : \mathtt{real} \otimes \mathtt{real} \vdash 
                & \> \text{pm} \> x \> \text{to} \> x_1 \otimes x_2 \> 
                \\
                & \> \mathtt{seq}_n(x_1) \>
                \mathtt{jump}_1 \dots \mathtt{jump}_1
                \\
                & \> \otimes
                \\
                & \> \mathtt{seq}_n(x_2) \>
                \mathtt{jump}_1 \dots \mathtt{jump}_1
                \\
                & =_{ n \left ( \frac{2\epsilon}{1 + \epsilon} + \frac{2\delta}{1 + \delta} \right ) }
                \\
                & \> \text{pm} \> x \> \text{to} \> x_1 \otimes x_2 \> 
                \\
                & \> \mathtt{seq}_n(x_1) \>
                \mathtt{jump}_{1+\epsilon} \dots \mathtt{jump}_{1+\epsilon}
                \\
                & \> \otimes
                \\
                & \> \mathtt{seq}_n(x_2) \>
                \mathtt{jump}_{1+\delta} \dots \mathtt{jump}_{1+\delta} : \mathtt{real} \otimes \mathtt{real}
        \end{align*}

\end{frame}
