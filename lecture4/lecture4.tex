\section{Theories, soundness, and completeness}

\begin{frame}{Theory}

        \begin{definition}[Signature]
                Set of basic types + operation symbols
                $f : \typeA_1, \dots, \typeA_n \to \typeB$
        \end{definition}


        \bigskip
        \pause
        \begin{definition}[Theory]
                A signature + a set of basic metric equations (axioms)
        \end{definition}


        \bigskip
        \pause
        \begin{definition}[Theorem]
                A metric equation that is provable in a given theory
        \end{definition}
\end{frame}

\begin{frame}{Example -- wait calls}

        \begin{itemize}
                \item A single basic type $X$
                        \\[7pt]
                \item countable set of operations $\left \{ \text{wait}_n : X \to X \mid n \in \Nats \right \}$
                        \\[7pt]
                \item the following axioms
        \end{itemize}
        \vspace{-0.4cm}
        \begin{align*}
                & \left \{ \text{wait}_n(x) =_{| n - m|} \text{wait}_m(x) \mid n,m \in \Nats \right \}
                \\[5pt]
                & \left \{ \text{wait}_n(\text{wait}_m(x)) =_0 \text{wait}_{n+m}(x) \mid n,m \in \Nats \right \}
                \\[5pt]
                & \left \{ \text{wait}_0(x) =_0 x \right \} 
        \end{align*}
\end{frame}

\begin{frame}{Example -- wait calls}
        We can then reason metrically, eg
        \[
                \infer{
                        \lambda x. \, x =_\epsilon \lambda x. \, \text{wait}_\epsilon(x)
                }{
                        \begin{aligned}
                                x 
                                & =_0 \text{wait}_0(x) 
                                \\
                                & =_\epsilon \text{wait}_\epsilon(x)
                        \end{aligned}
                }
        \]

        \pause
        \bigskip
        But is this correct ?
\end{frame}

\begin{frame}{Soundness}

        Pick a theory and any associated theorem $t =_q s$
        
        Pick an interpretation $\sem{-}$ 

        If all axioms are satisfied by the interpretation then
        \[
                d(\sem{t},\sem{s}) \leq q
        \]
                
        \pause
        \bigskip
        Proof sketch: routine induction over proof derivation trees
\end{frame}


\begin{frame}{Completeness}

        Pick a theory $\mathcal{T}$ and suppose all models satisfy
        \[
                d(\sem{t},\sem{s}) \leq q
        \]

        \medskip
        For every \alert{approximation} $q + \epsilon > q$
        \[
                t =_{q + \epsilon} s \text{ is a theorem }
        \]

        \pause
        \bigskip
        Proof idea: build a metrically enriched smc category
        from $\mathcal{T}$

        Then show $d(t,s) \leq q \Longrightarrow t =_{q + \epsilon} s$ is a theorem
\end{frame}

\section{Extensions}

\begin{frame}{Back to linearity}

        Linearity too strict ?

        \pause
        \medskip
        Not a problem !

        \pause
        \medskip
        Boost expressivity \alert{but} use the typing system to track it, eg
        \[
                x :\, !_n \, \typeA \vdash t : \typeB
                \hspace{1cm}
                \text{($x$ can be used $n$-times)}
        \]
\end{frame}

\begin{frame}{Other type constructs -- conditionals (disjunction)}

        \[
                \infer{
                        \Gamma, \Delta \vdash
                        \text{case} \>  t \> \{ \, x \Rightarrow s ; \> y \Rightarrow u \} 
                        : \typeC
                }{
                        \Gamma \vdash t : \typeA \oplus \typeB
                        \qquad
                        \Delta, x : \typeA \vdash s : \typeC
                        \qquad
                        \Delta, y : \typeB \vdash u : \typeC
                }
        \]

        \pause
        \bigskip
        \[
                \infer{
                        \text{case} \>  t \> \{ \, x \Rightarrow s ; \> y \Rightarrow u \, \} 
                        =_{q \vee r}
                        \text{case} \>  t \> \{ \, x \Rightarrow v ; \> y \Rightarrow w \, \} 
                }{
                        s =_q v \qquad u =_r w
                }
        \]
\end{frame}
